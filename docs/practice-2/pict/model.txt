status:            ACTIVE, INACTIVE, BLOCKED, EXPIRED
amount:            1, 15000000, 15000001
dailyLimit:        15000000, 30000000
monthlyLimit:      15000000, 300000000
availableBalance:  0, 15000000, 100000000
expiryDate:        0826, 0926, 1026
terminalType:      POS, ATM, ECOM
mcc:               5411, 5812, 5732

# Месячный лимит не может быть меньше дневного.
IF [dailyLimit] = 30000000 THEN [monthlyLimit] = 300000000;

IF [status] = "INACTIVE" THEN [amount] = 1 AND [dailyLimit] = 15000000
    AND [monthlyLimit] = 15000000 AND [availableBalance] = 100000000
    AND [expiryDate] = 1026;
IF [status] = "BLOCKED" THEN [amount] = 1 AND [dailyLimit] = 15000000
    AND [monthlyLimit] = 15000000 AND [availableBalance] = 100000000
    AND [expiryDate] = 1026;
IF [status] = "EXPIRED" THEN [amount] = 1 AND [dailyLimit] = 15000000
    AND [monthlyLimit] = 15000000 AND [availableBalance] = 100000000
    AND [expiryDate] = 826;

# После отказа по expiry лимиты и баланс не проверяются.
IF [expiryDate] = 826 THEN [amount] = 1 AND [dailyLimit] = 15000000
    AND [monthlyLimit] = 15000000 AND [availableBalance] = 100000000;
